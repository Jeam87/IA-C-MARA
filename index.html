import cv2 cap = cv2.VideoCapture(0) ret, frame = cap.read() if ret: cv2.imwrite('foto_actual.jpg', frame) cap.release() cv2.destroyAllWindows() import cv2 # Cargar modelo preentrenado para super resolución sr = cv2.dnn_superres.DnnSuperResImpl_create() path = "EDSR_x3.pb" # Modelo descargado previamente sr.readModel(path) sr.setModel("edsr", 3) # Leer la imagen img = cv2.imread('foto_actual.jpg') # Mejorar resolución result = sr.upsample(img) cv2.imwrite('foto_mejorada.jpg', result) import cv2 import numpy as np img = cv2.imread('foto_mejorada.jpg') # Convertir a YUV img_yuv = cv2.cvtColor(img, cv2.COLOR_BGR2YUV) # Equalización de histograma en luminancia img_yuv[:,:,0] = cv2.equalizeHist(img_yuv[:,:,0]) # Convertir de nuevo a BGR img_eq = cv2.cvtColor(img_yuv, cv2.COLOR_YUV2BGR) cv2.imwrite('foto_con_mejor_color.jpg', img_eq) import cv2 img = cv2.imread('foto_con_mejor_color.jpg') denoised = cv2.fastNlMeansDenoisingColored(img, None, h=10, hColor=10, templateWindowSize=7, searchWindowSize=21) cv2.imwrite('foto_final.jpg', denoised)
